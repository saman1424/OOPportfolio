package week8;

// Interface
interface Encryptable {
    void encrypt();
    String getEncrypted();
}

// Secret class
class Secret implements Encryptable {
    private String text;
    private String encrypted;

    public Secret(String text) {
        this.text = text;
    }

    public void encrypt() {
        encrypted = "";
        for (int i = 0; i < text.length(); i++) {
            encrypted += (char)(text.charAt(i) + 3);
        }
    }

    public String getEncrypted() {
        return encrypted;
    }
}

// Password class
class Password implements Encryptable {
    private String text;
    private String encrypted;

    public Password(String text) {
        this.text = text;
    }

    public void encrypt() {
        encrypted = "";
        for (int i = text.length() - 1; i >= 0; i--) {
            encrypted += text.charAt(i);
        }
    }

    public String getEncrypted() {
        return encrypted;
    }
}
