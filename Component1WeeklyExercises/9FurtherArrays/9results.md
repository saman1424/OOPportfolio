package Histogram;

	import java.util.Scanner;

	public class Histogram {
	    public static void main(String[] args) {
	        Scanner sc = new Scanner(System.in);
	        int[] count = new int[10];

	        System.out.println("Enter numbers (1–100), enter 0 to stop:");

	        while (true) {
	            int num = sc.nextInt();
	            if (num == 0) break;

	            if (num >= 1 && num <= 100) {
	                count[(num - 1) / 10]++;
	            } else {
	                System.out.println("Invalid number!");
	            }
	        }

	        // Print histogram
	        for (int i = 0; i < 10; i++) {
	            int start = i * 10 + 1;
	            int end = (i + 1) * 10;

	            System.out.printf("%2d - %3d | ", start, end);

	            for (int j = 0; j < count[i]; j++) {
	                System.out.print("*");
	            }
	            System.out.println();
	        }
	    }
	}

package Account;

import java.util.Scanner;

public class Bankdriver {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        Account[] accounts = new Account[30];
        int count = 0;

        while (true) {
            System.out.println("\n1.Create 2.Deposit 3.Withdraw 4.Add Interest 5.Show 0.Exit");
            int choice = sc.nextInt();

            if (choice == 0) break;

            switch (choice) {
                case 1:
                    if (count >= accounts.length) {
                        System.out.println("Account limit reached!");
                        break;
                    }
                    System.out.print("Name: ");
                    String name = sc.next();
                    System.out.print("Initial balance: ");
                    double bal = sc.nextDouble();
                    accounts[count++] = new Account(name, bal);
                    System.out.println("Account created at index " + (count - 1));
                    break;

                case 2:
                    System.out.print("Account index: ");
                    int i = sc.nextInt();
                    if (i >= 0 && i < count) {
                        System.out.print("Amount: ");
                        accounts[i].deposit(sc.nextDouble());
                    } else {
                        System.out.println("Invalid account index!");
                    }
                    break;

                case 3:
                    System.out.print("Account index: ");
                    int j = sc.nextInt();
                    if (j >= 0 && j < count) {
                        System.out.print("Amount: ");
                        accounts[j].withdraw(sc.nextDouble());
                    } else {
                        System.out.println("Invalid account index!");
                    }
                    break;

                case 4:
                    for (int k = 0; k < count; k++) {
                        accounts[k].addInterest();
                    }
                    System.out.println("Interest added to all accounts.");
                    break;

                case 5:
                    for (int k = 0; k < count; k++) {
                        System.out.print("Index " + k + ": ");
                        accounts[k].display();
                    }
                    break;

                default:
                    System.out.println("Invalid choice!");
            }
        }

        sc.close();
    }
}

package Account;

public class Account {
    private String name;
    private double balance;

    public Account(String name, double balance) {
        this.name = name;
        this.balance = balance;
    }

    public void deposit(double amount) {
        if (amount > 0) {
            balance += amount;
            System.out.println("Deposited successfully!");
        }
    }

    public void withdraw(double amount) {
        if (amount > 0 && amount <= balance) {
            balance -= amount;
            System.out.println("Withdraw successful!");
        } else {
            System.out.println("Insufficient balance!");
        }
    }

    public void addInterest() {
        balance += balance * 0.05; // 5% interest
    }

    public void display() {
        System.out.println("Name: " + name + ", Balance: " + balance);
    }
}

package postcode;
	import java.util.Scanner;

	public class PostCodeApp {
	    public static void main(String[] args) {
	        Scanner sc = new Scanner(System.in);
	        Person[] people = new Person[25];
	        int count = 0;

	        System.out.println("Enter FirstName LastName PostalCode (max 25):");

	        while (count < 25) {
	            String f = sc.next();
	            String l = sc.next();
	            String p = sc.next();

	            people[count++] = new Person(f, l, p);

	            System.out.print("Continue? (yes/no): ");
	            if (sc.next().equalsIgnoreCase("no"))
	                break;
	        }

	        System.out.println("\nStored Data:");
	        for (int i = 0; i < count; i++) {
	            people[i].display();
	        }
	    }
	}

package Quiztime;

import java.util.Scanner;

class Question {

    private String question;
    private String answer;

    public Question(String q, String a) {
        question = q;
        answer = a;
    }

    public boolean ask(Scanner sc) {

        System.out.println(question);

        String user = sc.nextLine();

        return user.equalsIgnoreCase(answer);
    }
}
```
```
package Quiztime;

import java.util.Scanner;

public class Quiz1 {

    private Question[] questions = new Question[25];
    private int count = 0;

    public void add(Question q) {
        questions[count++] = q;
    }

    public void giveQuiz() {

        Scanner sc = new Scanner(System.in);

        int score = 0;

        for (int i = 0; i < count; i++) {

            if (questions[i].ask(sc))
                score++;
        }

        System.out.println("Score: " + score + "/" + count);

        sc.close();
    }
}
