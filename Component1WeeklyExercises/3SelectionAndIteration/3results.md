package Salary;

import java.util.Scanner;
import java.text.NumberFormat;

public class Salary {
    public static void main(String[] args) {
        try (Scanner scan = new Scanner(System.in)) {
			NumberFormat money = NumberFormat.getCurrencyInstance();

			System.out.print("Enter salary: ");
			double salary = scan.nextDouble();
			System.out.print("Enter rating (Excellent, Good, Poor): ");
			String rating = scan.next().toLowerCase();

			// Assign percentage based on rating
			double percentage = switch (rating) {
			    case "excellent" -> 0.06;
			    case "good"      -> 0.04;
			    case "poor"      -> 0.015;
			    default          -> 0.0;
			};

			double raise = salary * percentage;
			
			System.out.println("\nRaise: " + money.format(raise));
			System.out.println("Total: " + money.format(salary + raise));
		}
    }
}

package Rock;

import java.util.Scanner;
import java.util.Random;

public class Rock {
    public static void main(String[] args) {
        try (Scanner scan = new Scanner(System.in)) {
			String[] plays = {"R", "P", "S"};
			
			System.out.print("Enter R, P, or S: ");
			String user = scan.next().toUpperCase();
			String cpu = plays[new Random().nextInt(3)];

			System.out.println("Computer played: " + cpu);

			if (user.equals(cpu)) {
			    System.out.println("It's a tie!");
			} else if ((user.equals("R") && cpu.equals("S")) || 
			           (user.equals("P") && cpu.equals("R")) || 
			           (user.equals("S") && cpu.equals("P"))) {
			    System.out.println("You win!");
			} else {
			    System.out.println("Computer wins!");
			}
		}
    }
}

package StringReverser;

import java.util.Scanner;

public class StringReverser {
    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);
        System.out.println("Enter a sentence:");
        String input = scan.nextLine();
        
        Scanner wordScanner = new Scanner(input);
        while (wordScanner.hasNext()) {
            String word = wordScanner.next();
            String reversedWord = "";
            for (int i = word.length() - 1; i >= 0; i--) {
                reversedWord += word.charAt(i);
            }
            System.out.print(reversedWord + " ");
        }
        wordScanner.close();
    }
}

package PunctuationCounter;

public class PunctuationCounter {
    public static void main(String[] args) {
        String text = "Mary had a little lamb, her fleece was as white as snow, and everywhere Mary went, the lamb was sure to go. -that was a nice poem- the end. ";
        int commas = 0, periods = 0, dashes = 0;

        for (int i = 0; i < text.length(); i++) {
            char c = text.charAt(i);
            if (c == ',') commas++;
            else if (c == '.') periods++;
            else if (c == '-') dashes++;
        }

        System.out.println("Punctuation Mark\tOccurrences");
        System.out.println("----------------\t-----------");
        System.out.println("Comma (,)       \t" + commas);
        System.out.println("Period (.)      \t" + periods);
        System.out.println("Dash (-)        \t" + dashes);
    }
}
