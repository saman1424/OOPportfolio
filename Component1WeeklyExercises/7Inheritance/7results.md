package inheritance;

class Employee {
 protected int empId;
 protected String name;

 public Employee(int empId, String name) {
     this.empId = empId;
     this.name = name;
 }
}

//Doctor class
class Doctor extends Employee {
 private String specialization;

 public Doctor(int empId, String name, String specialization) {
     super(empId, name);
     this.specialization = specialization;
 }

 public void specialize() {
     System.out.println("Doctor Emp#" + empId + " specializes in " + specialization);
 }

 public void diagnose() {
     System.out.println("Doctor Emp#" + empId + " is diagnosing a patient");
 }
}

//Nurse class
class Nurse extends Employee {
 private int patientCount;

 public Nurse(int empId, String name, int patientCount) {
     super(empId, name);
     this.patientCount = patientCount;
 }

 public void attendPatients() {
     System.out.println("Nurse Emp#" + empId + " has " + patientCount + " patients");
 }

 public void giveInjection() {
     System.out.println("Nurse Emp#" + empId + " is giving an injection");
 }
}

package inheritance;

class PlayerStats {
 protected String playerName;
 protected int age;
 protected String team;
 protected int gamesPlayed;

 public PlayerStats(String playerName, int age, String team, int gamesPlayed) {
     this.playerName = playerName;
     this.age = age;
     this.team = team;
     this.gamesPlayed = gamesPlayed;
 }

 public void displayInfo() {
     System.out.println("Player: " + playerName + " | Age: " + age +
                        " | Team: " + team + " | Games Played: " + gamesPlayed);
 }
}

//FootballStats subclass
class FootballStats extends PlayerStats {
 private int goals;
 private int assists;

 public FootballStats(String playerName, int age, String team, int gamesPlayed,
                      int goals, int assists) {
     super(playerName, age, team, gamesPlayed);
     this.goals = goals;
     this.assists = assists;
 }

 public void displayStats() {
     displayInfo();
     System.out.println("  Goals: " + goals + " | Assists: " + assists);
 }

 public void scoreGoal() {
     goals++;
     System.out.println(playerName + " scored a goal! Total goals: " + goals);
 }
}

//CricketStats subclass
class CricketStats extends PlayerStats {
 private int runs;
 private int wickets;

 public CricketStats(String playerName, int age, String team, int gamesPlayed,
                     int runs, int wickets) {
     super(playerName, age, team, gamesPlayed);
     this.runs = runs;
     this.wickets = wickets;
 }

 public void displayStats() {
     displayInfo();
     System.out.println("  Runs: " + runs + " | Wickets: " + wickets);
 }

 public void scoreRuns(int r) {
     runs += r;
     System.out.println(playerName + " scored " + r + " runs! Total runs: " + runs);
 }
}

//CyclingStats subclass
class CyclingStats extends PlayerStats {
 private double totalKm;
 private int racesWon;

 public CyclingStats(String playerName, int age, String team, int gamesPlayed,
                     double totalKm, int racesWon) {
     super(playerName, age, team, gamesPlayed);
     this.totalKm = totalKm;
     this.racesWon = racesWon;
 }

 public void displayStats() {
     displayInfo();
     System.out.println("  Total KM: " + totalKm + " | Races Won: " + racesWon);
 }

 public void completeRace(double km) {
     totalKm += km;
     System.out.println(playerName + " completed a " + km + "km race! Total KM: " + totalKm);
 }
}
